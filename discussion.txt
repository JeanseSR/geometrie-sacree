from __future__ import annotations

from figures.figure import Figure

from geometrie.cercle import Cercle
from geometrie.couronne import Couronne


class EpaisseurTrait:
    """
    Transforme une figure géométrique en figure
    fabriquable.

    V2 :
        Cercle -> Couronne

    Les autres objets sont simplement copiés.
    """

    def __init__(
        self,
        largeur: float
    ):

        if largeur <= 0:
            raise ValueError(
                "La largeur du trait doit être strictement positive."
            )

        self.largeur = largeur

    # ==========================================================
    # API PUBLIQUE
    # ==========================================================

    def appliquer(
        self,
        figure: Figure
    ) -> Figure:
        """
        Retourne une nouvelle Figure adaptée
        à la fabrication.
        """

        resultat = Figure()

        for objet in figure:

            if isinstance(objet, Cercle):

                resultat.ajouter(
                    self._creer_couronne(objet)
                )

            else:

                # Temporairement on recopie
                # les autres objets.
                resultat.ajouter(objet)

        return resultat

    # ==========================================================
    # PRIVÉ
    # ==========================================================

    def _creer_couronne(
        self,
        cercle: Cercle
    ) -> Couronne:

        demi = self.largeur / 2

        return Couronne(

            centre=cercle.centre,

            rayon_interieur=max(
                0,
                cercle.rayon - demi
            ),

            rayon_exterieur=(
                cercle.rayon + demi
            )

        )
